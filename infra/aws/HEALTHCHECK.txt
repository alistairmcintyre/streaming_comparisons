SQL
#
# (b) is the bronze job emitting records into the sink?
JID=$(k -n flink exec deploy/fluss-flink -- curl -s localhost:8081/jobs/overview \
      | python3 -c "import json,sys;print([j['jid'] for j in json.load(sys.stdin)['jobs'] if 'trades' in j['name']][0])")

── 8b2. ENRICHING GOLD (country/tier live in the dimension, not the fact) ─────
# gold.open_positions holds POSITION FACTS only — 9 identical columns in all five
# engines. country/tier are account attributes and were removed from gold: on a
# current-state row their value had no defensible meaning (it was whatever the last
# batch to touch the row happened to see), and denormalising them cost a full
# silver.accounts rescan every 15s in Spark and a second copy of the book in Flink
# join state. Join at read time instead:
#
#   SELECT p.*, a.country, a.tier
#   FROM gold.open_positions_delta p
#   LEFT JOIN silver.accounts_delta a USING (account_id)
#
# LEFT, never INNER: trades and accounts are independent CDC streams, so a fill can
# land before its account row. INNER would silently drop that position and make the
# book's row count depend on dimension timing.

── 9. LATENCY + GRAFANA ───────────────────────────────────────────────────────
k -n kafka exec trades-dual-role-0 -- bin/kafka-get-offsets.sh \
  --bootstrap-server localhost:9092 --topic pipeline_latency
# expect: offsets > 0. Pipelines emitting: {delta,iceberg,hudi}-{bronze,gold},
# {paimon,fluss}-{bronze|silver,gold}. This feeds the LIVE dashboard only.
#
# The HEADLINE processing-delay number does not come from here — it is read out of
# the gold tables themselves as (commit_ts - last_updated_at), which every engine
# computes identically and which cannot silently vanish if the emit chain breaks:
./infra/aws/scripts/query-lake-local.sh "SELECT engine, approx_quantile(delay_ms,0.5) p50
  FROM (SELECT 'delta' engine, epoch_ms(commit_ts)-epoch_ms(last_updated_at) delay_ms
        FROM delta_gold_positions) GROUP BY 1"
# or read processing_delay.csv from the run's benchmarks/ prefix after teardown.
